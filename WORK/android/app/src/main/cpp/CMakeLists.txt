cmake_minimum_required(VERSION 3.22.1)
project(uno_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_ROOT}/include/llama.h")
    message(FATAL_ERROR "Pinned llama.cpp submodule absent: git submodule update --init --recursive")
endif()
add_subdirectory("${LLAMA_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/upstream-llama" EXCLUDE_FROM_ALL)
add_library(uno_llama SHARED uno_llama.cpp)
target_include_directories(uno_llama PRIVATE "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/ggml/include")
target_link_libraries(uno_llama PRIVATE llama log android)
